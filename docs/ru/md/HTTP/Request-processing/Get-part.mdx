  | КоличествоБайт | - | Число | &#x2714; | Размер получаемых данных в байтах |

```bsl title="Пример использования для 1С:Предприятие/OneScript"
    URL = "https://bin.openintegrations.dev";
    URL = URL + "/range/100";

    // Получаем байты 0-9
    НачальнаяПозиция = 0;
    ЧислоБайт        = 10;

    Результат = OPI_ЗапросыHTTP.НовыйЗапрос()
        .Инициализировать(URL)
        .ПолучитьЧасть(НачальнаяПозиция, ЧислоБайт) // <---
        .ВернутьОтвет();
```
